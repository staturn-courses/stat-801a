#Problem 1 - lead poisoning

control<-c(5.4,6.2,3.1,3.8,6.5,5.8,6.4,4.5,4.9,4.0)
exp<-c(8.8,9.5,10.6,9.6,7.5,6.9,7.4,6.5,10.5,8.3)

mean(control)
mean(exp)

var(control)
var(exp)

#Solution to part (e)
t.test(control,exp,mu=0,alternative="less",var.equal=TRUE)

#Solution to part (f)
t.test(control,exp,mu=0,alternative="two.sided",var.equal=TRUE,conf.level=0.90)

#Problem 2 - polyester decay

two<-c(118,135,127,120,129,123,115,130)
sixteen<-c(125,98,113,136,110,121,118,105)

#Solution to part (d)

par(mfrow=c(1,2))
qqnorm(two,main="2")
qqnorm(sixteen,main="16")

par(mfrow=c(1,2))
boxplot(two,main="2")
boxplot(sixteen,main="16")

#Solution to part (e)

t.test(two,sixteen,mu=0,alternative="greater")

#Solution to part (f)

t.test(two,sixteen,mu=0,alternative="two.sided",conf.level=0.9)
