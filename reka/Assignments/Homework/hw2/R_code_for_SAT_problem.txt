#Histogram of the data
# Be careful! Since our data is a matrix, we need to specify which column we are talking about

#Plotting the histogram for the first 100 rows and first column of the data (which in our case is the first 100 observations)
hist(data[1:100,1])

#Summary statistics of the data
summary(data[,1])
